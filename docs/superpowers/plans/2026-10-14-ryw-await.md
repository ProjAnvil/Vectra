	case <-time.After(time.Second):

	}

func TestWaitPublishedWakesOnPublish(t *testing.T) {
	f := newFSM(time.Hour)
	defer f.Close()
	const want = 7
	done := make(chan uint64, 1)
	go func() {
		ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
		defer cancel()
		done <- f.WaitPublished(ctx, want)
	}()
	time.Sleep(50 * time.Millisecond) // let the waiter arm (result assertion stays deterministic)
	f.mu.Lock()
	f.live.AddNode("a", "S", nil)
	f.mu.Unlock()
	f.appliedIndex.Store(want)
	f.dirty = true
	f.sealNow()
	select {
	case got := <-done:
		if got < want {
			t.Fatalf("WaitPublished = %d, want >= %d", got, want)
		}
	case <-time.After(time.Second):
		t.Fatal("waiter was not woken by publish")
	}
}

	defer cancel()

		t.Fatalf("got = %d, want current seq 0", got)

- [ ] **Step 3:** 实现——fsm.go：
  - struct 加 `pubNotify atomic.Pointer[chan struct{}]`；`newFSM` 里 `ch := make(chan struct{}); f.pubNotify.Store(&ch)`；
  - `publishOnce` 的 `if version > f.publishedVer { ... f.published.Store(...); ... }` 块末尾（仍在 pubMu 内）：`old := f.pubNotify.Swap(&newCh)`——先 `newCh := make(chan struct{})`，`close(*old)`；
  - 方法：

```go
// waitCh returns the current publish-notification channel; it is closed on
// the next successful publish.
func (f *fsm) waitCh() <-chan struct{} { return *f.pubNotify.Load() }

// WaitPublished blocks until publishedSeq >= minSeq or ctx ends, then
// returns the current publishedSeq. Lock-free: publishers close-and-swap
// pubNotify under pubMu; waiters double-check the seq around snapshotting
// the channel so a publish landing between check and arm is never missed.
func (f *fsm) WaitPublished(ctx context.Context, minSeq uint64) uint64 {
	for {
		if seq := f.PublishedSeq(); seq >= minSeq {
			return seq
		}
		ch := f.waitCh()
		if seq := f.PublishedSeq(); seq >= minSeq {
			return seq
		}
		select {
		case <-ch:
		case <-ctx.Done():
			return f.PublishedSeq()
		}
	}
}
```

```go
// await limits: server-side long-poll cap and the default when timeoutMs is
// omitted.
const (
	maxAwaitTimeout  = 30 * time.Second
	defaultAwaitWait = 5 * time.Second
)

func (n *Node) handleAwait(w http.ResponseWriter, r *http.Request) {
	seq, err := strconv.ParseUint(r.URL.Query().Get("seq"), 10, 64)
	if err != nil {
		writeErr(w, http.StatusBadRequest, "VALIDATION", "seq query parameter required (uint)")
		return
	}
	wait := defaultAwaitWait
	if s := r.URL.Query().Get("timeoutMs"); s != "" {
		ms, err := strconv.Atoi(s)
		if err != nil || ms < 0 {
			writeErr(w, http.StatusBadRequest, "VALIDATION", "timeoutMs must be a non-negative integer")
			return
		}
		wait = min(time.Duration(ms)*time.Millisecond, maxAwaitTimeout)
	}
	// No readiness gate: waiting for the first publish is legitimate (a
	// Restore's first publish carries a seq and wakes waiters).
	ctx, cancel := context.WithTimeout(r.Context(), wait)
	defer cancel()
	got := n.fsm.WaitPublished(ctx, seq)
	w.Header().Set("X-Vectra-Snapshot", strconv.FormatUint(got, 10))
	writeJSON(w, http.StatusOK, map[string]any{"publishedSeq": got, "reached": got >= seq})
}
```

（`context` import；handler 不计 queryCount——不是图查询。）

```go
// awaitResponse mirrors GET /v1/await.
type awaitResponse struct {
	PublishedSeq float64 `json:"publishedSeq"`
	Reached      bool    `json:"reached"`
}

func waitForPublish(c *http.Client, base string, index uint64, waitCap time.Duration) {
	deadline := time.Now().Add(waitCap)
	for {
		remaining := time.Until(deadline)
		if remaining <= 0 {
			return
		}
		url := base + "/v1/await?seq=" + strconv.FormatUint(index, 10) +
			"&timeoutMs=" + strconv.Itoa(int(min(remaining, 5*time.Second).Milliseconds()))
		resp, err := c.Get(url)
		if err == nil {
			var out awaitResponse
			derr := json.NewDecoder(resp.Body).Decode(&out)
			resp.Body.Close()
			if derr == nil && out.Reached {
				return
			}
		}
		// Unsatisfied (timeout) or transport error: loop until the cap.
		if !time.Now().Before(deadline) {
			return
		}
	}
}
```
