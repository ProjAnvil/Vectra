```go
// Package csr 实现压缩稀疏行邻接结构：节点 n 的邻居存放在
// Targets[Offsets[n]:Offsets[n+1]]。全部为 uint32 平坦数组，无指针，
// GC 扫描成本与边数无关。
package csr

// Adjacency 是一份 CSR 邻接。EdgeIDs 与 Labels 与 Targets 平行，
// 可为 nil（Build 不带 payload 时）。
type Adjacency struct {
	Offsets []uint32 // len == numNodes+1
	Targets []uint32
	EdgeIDs []uint32
	Labels  []uint32
}

// Build 用计数排序从边表构建 CSR，同节点邻居保持插入序。
func Build(numNodes uint32, src, dst []uint32) Adjacency {
	return build(numNodes, src, dst, nil, nil)
}

// BuildPayloads 同 Build，且每个槽位携带对应的边 ID 与标签 ID。
func BuildPayloads(numNodes uint32, src, dst, edgeIDs, labels []uint32) Adjacency {
	return build(numNodes, src, dst, edgeIDs, labels)
}

func build(numNodes uint32, src, dst, edgeIDs, labels []uint32) Adjacency {
	a := Adjacency{Offsets: make([]uint32, int(numNodes)+1)}
	for _, s := range src {
		a.Offsets[s+1]++
	}
	for i := uint32(1); i <= numNodes; i++ {
		a.Offsets[i] += a.Offsets[i-1]
	}
	n := len(src)
	a.Targets = make([]uint32, n)
	if edgeIDs != nil {
		a.EdgeIDs = make([]uint32, n)
	}
	if labels != nil {
		a.Labels = make([]uint32, n)
	}
	cursor := make([]uint32, numNodes)
	copy(cursor, a.Offsets[:numNodes])
	for i, s := range src {
		pos := cursor[s]
		cursor[s]++
		a.Targets[pos] = dst[i]
		if a.EdgeIDs != nil {
			a.EdgeIDs[pos] = edgeIDs[i]
		}
		if a.Labels != nil {
			a.Labels[pos] = labels[i]
		}
	}
	return a
}

// Degree 返回节点 n 的出度。
func (a Adjacency) Degree(n uint32) uint32 {
	return a.Offsets[n+1] - a.Offsets[n]
}

// Neighbors 返回节点 n 的邻居视图（底层数组切片，调用方不得修改）。
func (a Adjacency) Neighbors(n uint32) []uint32 {
	return a.Targets[a.Offsets[n]:a.Offsets[n+1]]
}

// Row 返回节点 n 的整行：邻居、边 ID、标签三段平行视图。
func (a Adjacency) Row(n uint32) (targets, edgeIDs, labels []uint32) {
	lo, hi := a.Offsets[n], a.Offsets[n+1]
	return a.Targets[lo:hi], a.EdgeIDs[lo:hi], a.Labels[lo:hi]
}
```

```go
func normalizeSCC(sccOf []uint32) map[uint32][]uint32 {
	m := map[uint32][]uint32{}

		m[s] = append(m[s], uint32(v))

	if len(got) != len(want) {
		return false
	}
	seen := map[uint32]bool{}
	for _, v := range got {
		seen[v] = true
	}
	for _, w := range want {
		if !seen[w] {
			return false
		}
	}
	return true
```

```go
// Package dag 提供强连通分量（Tarjan 迭代版）、凝聚图构建与拓扑层级。
// 查询侧的"写入时压环"由本包支撑：Seal 时压缩 SCC，之后查询面对 DAG。
package dag

import "slices"

type frame struct {
	v  uint32
	ei uint32
}

// Tarjan 计算强连通分量。返回 sccOf（节点->分量 ID）与分量总数。
// 迭代实现：显式调用栈，深度不受 goroutine 栈限制。
func Tarjan(numNodes uint32, adj func(uint32) []uint32) (sccOf []uint32, count uint32) {
	const unvisited = ^uint32(0)
	index := make([]uint32, numNodes)
	for i := range index {
		index[i] = unvisited
	}
	low := make([]uint32, numNodes)
	onStack := make([]bool, numNodes)
	sccOf = make([]uint32, numNodes)
	var tarjanStack []uint32
	var callStack []frame
	var counter uint32

	for root := uint32(0); root < numNodes; root++ {
		if index[root] != unvisited {
			continue
		}
		index[root] = counter
		low[root] = counter
		counter++
		tarjanStack = append(tarjanStack, root)
		onStack[root] = true
		callStack = append(callStack, frame{v: root})
		for len(callStack) > 0 {
			f := &callStack[len(callStack)-1]
			nbrs := adj(f.v)
			if f.ei < uint32(len(nbrs)) {
				w := nbrs[f.ei]
				f.ei++
				if index[w] == unvisited {
					index[w] = counter
					low[w] = counter
					counter++
					tarjanStack = append(tarjanStack, w)
					onStack[w] = true
					callStack = append(callStack, frame{v: w})
				} else if onStack[w] && index[w] < low[f.v] {
					low[f.v] = index[w]
				}
				continue
			}
			// 节点 f.v 的所有邻居处理完毕
			if low[f.v] == index[f.v] {
				for {
					w := tarjanStack[len(tarjanStack)-1]
					tarjanStack = tarjanStack[:len(tarjanStack)-1]
					onStack[w] = false
					sccOf[w] = count
					if w == f.v {
						break
					}
				}
				count++
			}
			callStack = callStack[:len(callStack)-1]
			if len(callStack) > 0 {
				p := &callStack[len(callStack)-1]
				if low[f.v] < low[p.v] {
					low[p.v] = low[f.v]
				}
			}
		}
	}
	return sccOf, count
}

// Condensed 是凝聚图的 CSR 表示。Labels 平行于 Targets：该凝聚边继承的
// 原边标签（跨 SCC 且同标签的平行边去重后保留一条）。
type Condensed struct {
	Offsets []uint32 // len == sccCount+1
	Targets []uint32
	Labels  []uint32
}

func (c Condensed) Row(s uint32) (targets, labels []uint32) {
	lo, hi := c.Offsets[s], c.Offsets[s+1]
	return c.Targets[lo:hi], c.Labels[lo:hi]
}

type condEdge struct {
	src, dst, label uint32
}

// Condense 构建凝聚图。labelOf 返回与 adj(u) 平行的每条出边标签，
// 为 nil 时凝聚边不带标签。跨 SCC 边按 (src, dst, label) 三元组去重。
func Condense(numNodes uint32, sccOf []uint32, count uint32, adj func(uint32) []uint32, labelOf func(uint32) []uint32) Condensed {
	var edges []condEdge
	for u := uint32(0); u < numNodes; u++ {
		nbrs := adj(u)
		labs := labelOf(u)
		for i, v := range nbrs {
			if sccOf[u] == sccOf[v] {
				continue
			}
			e := condEdge{src: sccOf[u], dst: sccOf[v]}
			if labs != nil {
				e.label = labs[i]
			}
			edges = append(edges, e)
		}
	}
	slices.SortFunc(edges, func(a, b condEdge) int {
		if a.src != b.src {
			return int(a.src) - int(b.src)
		}
		if a.dst != b.dst {
			return int(a.dst) - int(b.dst)
		}
		return int(a.label) - int(b.label)
	})
	edges = slices.CompactFunc(edges, func(a, b condEdge) bool { return a == b })

	c := Condensed{Offsets: make([]uint32, count+1)}
	for _, e := range edges {
		c.Offsets[e.src+1]++
	}
	for i := uint32(1); i <= count; i++ {
		c.Offsets[i] += c.Offsets[i-1]
	}
	c.Targets = make([]uint32, len(edges))
	if labelOf != nil {
		c.Labels = make([]uint32, len(edges))
	}
	for i, e := range edges {
		c.Targets[i] = e.dst
		if c.Labels != nil {
			c.Labels[i] = e.label
		}
	}
	return c
}

// Levels 用 Kahn 拓扑排序计算凝聚图最长路径层级：入度 0 为 0 层，
// 其余为 1 + max(前驱层级)。DAG 上所有边严格升层。
func Levels(c Condensed) []uint32 {
	n := uint32(len(c.Offsets) - 1)
	level := make([]uint32, n)
	indeg := make([]uint32, n)
	for _, d := range c.Targets {
		indeg[d]++
	}
	queue := make([]uint32, 0, n)
	for s := uint32(0); s < n; s++ {
		if indeg[s] == 0 {
			queue = append(queue, s)
		}
	}
	for i := 0; i < len(queue); i++ {
		s := queue[i]
		tg, _ := c.Row(s)
		for _, d := range tg {
			if level[s]+1 > level[d] {
				level[d] = level[s] + 1
			}
			indeg[d]--
			if indeg[d] == 0 {
				queue = append(queue, d)
			}
		}
	}
	return level
}
```

```go
// Package vectra 是嵌入式内存图引擎：全内存无指针 CSR 邻接 + 写入侧
// SCC 压缩 + 层同步并行 BFS。面向"图永远小于单机内存"的企业依赖图谱
// 场景：批量导入、Seal 后并发只读、磁盘仅作快照恢复。
package vectra

import (
	"errors"

	"vectra/internal/csr"
	"vectra/internal/dag"
)

type NodeID uint32
type EdgeID uint32

var (
	ErrSealed       = errors.New("vectra: graph sealed, mutation rejected")
	ErrNotSealed    = errors.New("vectra: graph not sealed, query rejected")
	ErrNodeNotFound = errors.New("vectra: node key not found")
)

type Node struct {
	ID    NodeID
	Key   string
	Label string
	Props map[string]any
}

type Edge struct {
	ID    EdgeID
	Src   NodeID
	Dst   NodeID
	Label string
	Props map[string]any
}

type Stats struct {
	Nodes        int
	Edges        int
	SCCs         int
	CyclicNodes  int // 属于大小>1 的 SCC 的节点数（不含自环）
}

type edgeRec struct {
	src, dst   NodeID
	label      uint32
	props      map[string]any
}

type Graph struct {
	nodes  []Node
	byKey  map[string]NodeID
	edges  []edgeRec
	labels []string          // labelID -> name
	labIDs map[string]uint32
	sealed bool

	// Seal 产物（不可变，并发读免锁）
	csrF      csr.Adjacency // 正向
	csrR      csr.Adjacency // 反向
	sccOf     []uint32
	sccCount  uint32
	cond      dag.Condensed
	sccLevel  []uint32
	sccNodeOff []uint32 // scc -> 成员区间
	sccNodes  []uint32
	selfLoop  []bool
}

func New() *Graph {
	return &Graph{byKey: map[string]NodeID{}, labIDs: map[string]uint32{}}
}

func (g *Graph) internLabel(name string) uint32 {
	if id, ok := g.labIDs[name]; ok {
		return id
	}
	id := uint32(len(g.labels))
	g.labels = append(g.labels, name)
	g.labIDs[name] = id
	return id
}

func copyProps(props map[string]any) map[string]any {
	if props == nil {
		return nil
	}
	m := make(map[string]any, len(props))
	for k, v := range props {
		m[k] = v
	}
	return m
}

func (g *Graph) AddNode(key, label string, props map[string]any) (NodeID, error) {
	if g.sealed {
		return 0, ErrSealed
	}
	if id, ok := g.byKey[key]; ok {
		g.nodes[id].Label = label
		g.nodes[id].Props = copyProps(props)
		return id, nil
	}
	id := NodeID(len(g.nodes))
	g.nodes = append(g.nodes, Node{ID: id, Key: key, Label: label, Props: copyProps(props)})
	g.byKey[key] = id
	return id, nil
}

func (g *Graph) AddEdge(srcKey, dstKey, label string, props map[string]any) (EdgeID, error) {
	if g.sealed {
		return 0, ErrSealed
	}
	src, ok := g.byKey[srcKey]
	if !ok {
		return 0, ErrNodeNotFound
	}
	dst, ok := g.byKey[dstKey]
	if !ok {
		return 0, ErrNodeNotFound
	}
	g.internLabel(label)
	g.edges = append(g.edges, edgeRec{src: src, dst: dst, label: g.labIDs[label], props: copyProps(props)})
	return EdgeID(len(g.edges) - 1), nil
}

func (g *Graph) NodeByKey(key string) (Node, bool) {
	id, ok := g.byKey[key]
	if !ok {
		return Node{}, false
	}
	return g.nodes[id], true
}

func (g *Graph) NodeByID(id NodeID) (Node, bool) {
	if int(id) >= len(g.nodes) {
		return Node{}, false
	}
	return g.nodes[id], true
}

func (g *Graph) EdgeByID(id EdgeID) (Edge, bool) {
	if int(id) >= len(g.edges) {
		return Edge{}, false
	}
	e := g.edges[id]
	return Edge{ID: id, Src: e.src, Dst: e.dst, Label: g.labels[e.label], Props: e.props}, true
}

func (g *Graph) NodeCount() int { return len(g.nodes) }
func (g *Graph) EdgeCount() int { return len(g.edges) }
func (g *Graph) Sealed() bool   { return g.sealed }

func (g *Graph) Stats() Stats {
	st := Stats{Nodes: len(g.nodes), Edges: len(g.edges)}
	if !g.sealed {
		return st
	}
	st.SCCs = int(g.sccCount)
	for s := uint32(0); s < g.sccCount; s++ {
		if g.sccNodeOff[s+1]-g.sccNodeOff[s] > 1 {
			st.CyclicNodes += int(g.sccNodeOff[s+1] - g.sccNodeOff[s])
		}
	}
	return st
}

// Seal 构建查询所需的全部不可变结构：正/反向 CSR、SCC 分量、
// 凝聚 DAG、拓扑层级、分量成员表。此后图只读、并发读免锁。
func (g *Graph) Seal() error {
	if g.sealed {
		return nil
	}
	n := uint32(len(g.nodes))
	e := len(g.edges)
	src := make([]uint32, e)
	dst := make([]uint32, e)
	eids := make([]uint32, e)
	lids := make([]uint32, e)
	for i, er := range g.edges {
		src[i] = uint32(er.src)
		dst[i] = uint32(er.dst)
		eids[i] = uint32(i)
		lids[i] = er.label
	}
	g.csrF = csr.BuildPayloads(n, src, dst, eids, lids)
	g.csrR = csr.BuildPayloads(n, dst, src, eids, lids)

	adjF := func(u uint32) []uint32 { return g.csrF.Neighbors(u) }
	labF := func(u uint32) []uint32 { return g.csrF.Labels[g.csrF.Offsets[u]:g.csrF.Offsets[u+1]] }

	g.sccOf, g.sccCount = dag.Tarjan(n, adjF)
	g.cond = dag.Condense(n, g.sccOf, g.sccCount, adjF, labF)
	g.sccLevel = dag.Levels(g.cond)

	// 分量成员表：计数排序按 sccOf 分组（组内保持节点 ID 升序——
	// 按 u 升序遍历天然有序）
	g.sccNodeOff = make([]uint32, g.sccCount+1)
	for _, s := range g.sccOf {
		g.sccNodeOff[s+1]++
	}
	for i := uint32(1); i <= g.sccCount; i++ {
		g.sccNodeOff[i] += g.sccNodeOff[i-1]
	}
	g.sccNodes = make([]uint32, n)
	cursor := make([]uint32, g.sccCount)
	copy(cursor, g.sccNodeOff[:g.sccCount])
	for u := uint32(0); u < n; u++ {
		s := g.sccOf[u]
		g.sccNodes[cursor[s]] = u
		cursor[s]++
	}

	g.selfLoop = make([]bool, n)
	for _, er := range g.edges {
		if er.src == er.dst {
			g.selfLoop[er.src] = true
		}
	}

	g.sealed = true
	return nil
}
```

```go
package vectra

import "sync"

// 位图缓冲池：BFS 的 visited 位图按字数复用，查询结束归还前清零。
type bitmapPool struct {
	p sync.Pool
}

func (bp *bitmapPool) acquire(words int) []uint32 {
	if v, ok := bp.p.Get().(*[]uint32); ok && len(*v) >= words {
		b := (*v)[:words]
		clear(b)
		return b
	}
	return make([]uint32, words)
}

func (bp *bitmapPool) release(b []uint32) {
	clear(b)
	bp.p.Put(&b)
}

var visitedPool bitmapPool

func wordsFor(n int) int { return (n + 31) / 32 }
```

```go
package vectra

type Direction uint8

const (
	DirOut Direction = iota

type BFSResult struct {
	Layers [][]NodeID
	Depths map[NodeID]uint32
}

func (r *BFSResult) Nodes() []NodeID {
	total := 0
	for _, l := range r.Layers {
		total += len(l)
	}
	out := make([]NodeID, 0, total)
	for _, l := range r.Layers {
		out = append(out, l...)
	}
	return out

func (r *BFSResult) Count() int {
	total := 0
	for _, l := range r.Layers {
		total += len(l)
	}
	return total

type bfsConfig struct {
	maxDepth    uint32
	labels      []uint32
	labelNames  []string
	direction   Direction
	parallelThr int
}

type BFSOption func(*bfsConfig)

func WithMaxDepth(d uint32) BFSOption      { return func(c *bfsConfig) { c.maxDepth = d } }
func WithDirection(d Direction) BFSOption  { return func(c *bfsConfig) { c.direction = d } }
func WithParallelThreshold(n int) BFSOption { return func(c *bfsConfig) { c.parallelThr = n } }

// WithLabels 只沿给定标签的边遍历。空列表 = 不过滤。
func WithLabels(names ...string) BFSOption {
	return func(c *bfsConfig) { c.labelNames = names }
}

func (g *Graph) adjacency(dir Direction) csrAdjacency {
	if dir == DirIn {
		return csrAdjacency{g.csrR}
	}
	return csrAdjacency{g.csrF}
}

// csrAdjacency 是 traverse 内部对 csr.Adjacency 的别名包装（避免导出 csr）。
type csrAdjacency struct {
	a interface {
		Degree(uint32) uint32
		Neighbors(uint32) []uint32
	}
}

// BFS 多源层同步广度优先。结果含每层成员与每节点深度（起点深度 0）。
// 层同步是并行化的基础：Task 5 在 frontier 达到阈值时切换并行展开。
func (g *Graph) BFS(starts []NodeID, opts ...BFSOption) (*BFSResult, error) {
	if !g.sealed {
		return nil, ErrNotSealed
	}
	cfg := bfsConfig{maxDepth: ^uint32(0), parallelThr: 1024}
	for _, o := range opts {
		o(&cfg)
	}
	var mask []bool
	if len(cfg.labelNames) > 0 {
		mask = make([]bool, len(g.labels))
		any := false
		for _, n := range cfg.labelNames {
			if id, ok := g.labIDs[n]; ok {
				mask[id] = true
				any = true
			}
		}
		if !any {
			mask = nil // 标签都不存在：与全部过滤等价，返回仅起点
			mask = make([]bool, len(g.labels)) // 全 false = 什么都不通过
		}
	}
	adj := g.adjacency(cfg.direction)

	n := uint32(len(g.nodes))
	for _, s := range starts {
		if uint32(s) >= n {
			return nil, ErrNodeNotFound
		}
	}
	visited := visitedPool.acquire(wordsFor(int(n)))
	defer visitedPool.release(visited)

	res := &BFSResult{Depths: make(map[NodeID]uint32, 16)}
	frontier := make([]NodeID, 0, len(starts))
	for _, s := range starts {
		if !testAndSet(visited, uint32(s)) {
			frontier = append(frontier, s)
			res.Depths[s] = 0
		}
	}
	res.Layers = append(res.Layers, frontier)
	for depth := uint32(1); depth <= cfg.maxDepth && len(frontier) > 0; depth++ {
		var next []NodeID
		if len(frontier) >= cfg.parallelThr {
			next = g.expandLayerParallel(adj, mask, visited, frontier, n)
		} else {
			next = g.expandLayerSerial(adj, mask, visited, frontier)
		}
		for _, v := range next {
			res.Depths[v] = depth
		}
		res.Layers = append(res.Layers, next)
		frontier = next
	}
	return res, nil
}

func testAndSet(words []uint32, v uint32) bool {
	w := &words[v>>5]
	bit := uint32(1) << (v & 31)
	if *w&bit != 0 {
		return true
	}
	*w |= bit
	return false
}

func (g *Graph) expandLayerSerial(adj csrAdjacency, mask []bool, visited []uint32, frontier []NodeID) []NodeID {
	var next []NodeID
	for _, u := range frontier {
		next = appendNeighbors(next, adj, mask, visited, uint32(u), false)
	}
	return next
}

// appendNeighbors 展开 u 的邻接行，testAndSet 去重后追加新节点。
// atomicOr=true 时用原子操作（并行展开路径）。
func appendNeighbors(next []NodeID, adj csrAdjacency, mask []bool, visited []uint32, u uint32, atomicOr bool) []NodeID {
	lo, hi := g_offsets(adj, u)
	for s := lo; s < hi; s++ {
		if mask != nil && !mask[g_label(adj, s)] {
			continue
		}
		v := g_target(adj, s)
		if atomicOr {
			w := &visited[v>>5]
			bit := uint32(1) << (v & 31)
			if atomic.OrUint32(w, bit)&bit != 0 {
				continue // 已被其他 worker 标记
			}
		} else if testAndSet(visited, v) {
			continue
		}
		next = append(next, NodeID(v))
	}
	return next
}
```

```go
import (
	"runtime"
	"sync"
	"sync/atomic"
)

const parallelChunk = 256 // 每个 chunk 的最小 frontier 大小

// expandLayerParallel 将 frontier 切块分给多个 goroutine 并行展开，
// 去重靠原子 Or 位图：某 worker 置位前该位为 0 才拥有该节点。
// 层同步 = morsel 驱动：每层的 frontier 是扁平数组，天然并行单元。
func (g *Graph) expandLayerParallel(adj csrAdjacency, mask []bool, visited []uint32, frontier []NodeID, numNodes uint32) []NodeID {
	workers := runtime.GOMAXPROCS(0)
	if m := (len(frontier) + parallelChunk - 1) / parallelChunk; m < workers {
		workers = m
	}
	if workers <= 1 {
		return g.expandLayerSerial(adj, mask, visited, frontier)
	}
	chunk := (len(frontier) + workers - 1) / workers
	locals := make([][]NodeID, workers)
	var wg sync.WaitGroup
	for w := 0; w < workers; w++ {
		lo := w * chunk
		if lo >= len(frontier) {
			break
		}
		hi := min(lo+chunk, len(frontier))
		wg.Add(1)
		go func(w, lo, hi int) {
			defer wg.Done()
			local := make([]NodeID, 0, (hi-lo)*2)
			for _, u := range frontier[lo:hi] {
				local = appendNeighbors(local, adj, mask, visited, uint32(u), true)
			}
			locals[w] = local
		}(w, lo, hi)
	}
	wg.Wait()
	var next []NodeID
	for _, l := range locals {
		next = append(next, l...)
	}
	return next
}
```

```go
// Reachable 返回从 start 出发（含自身）可达的全部节点，按 NodeID 升序。
// 在 SCC 凝聚 DAG 上做超节点 BFS：查询面对的图无环，访问位图每位对应
// 一个分量，环检测成本为零——这是写入侧压环换来的查询侧红利。
func (g *Graph) ReachableIDs(start NodeID, opts ...ReachOption) ([]NodeID, error) {
	if !g.sealed {
		return nil, ErrNotSealed
	}
	n := uint32(len(g.nodes))
	if uint32(start) >= n {
		return nil, ErrNodeNotFound
	}
	cfg := reachConfig{direction: DirOut}
	for _, o := range opts {
		o(&cfg)
	}
	cond := g.cond
	if cfg.direction == DirIn {
		cond = g.condR
	}
	var mask []bool
	if len(cfg.labelNames) > 0 {
		mask = make([]bool, len(g.labels))
		for _, name := range cfg.labelNames {
			if id, ok := g.labIDs[name]; ok {
				mask[id] = true
			}
		}
	}
	nSCC := int(g.sccCount)
	seen := make([]bool, nSCC)
	startSCC := g.sccOf[start]
	seen[startSCC] = true
	frontier := []uint32{startSCC}
	for len(frontier) > 0 {
		var next []uint32
		for _, s := range frontier {
			tg, lb := cond.Row(s)
			for i, d := range tg {
				if mask != nil && !mask[lb[i]] {
					continue
				}
				if !seen[d] {
					seen[d] = true
					next = append(next, d)
				}
			}
		}
		frontier = next
	}
	out := make([]NodeID, 0, n)
	for s := uint32(0); s < g.sccCount; s++ {
		if !seen[s] {
			continue
		}
		for _, v := range g.sccNodes[g.sccNodeOff[s]:g.sccNodeOff[s+1]] {
			out = append(out, NodeID(v))
		}
	}
	return out, nil // sccNodes 按分量分组、组内 NodeID 升序 → 整体升序
}

func (g *Graph) Reachable(startKey string, opts ...ReachOption) ([]string, error) {
	node, ok := g.NodeByKey(startKey)
	if !ok {
		return nil, ErrNodeNotFound
	}
	ids, err := g.ReachableIDs(node.ID, opts...)
	if err != nil {
		return nil, err
	}
	out := make([]string, 0, len(ids))
	for _, id := range ids {
		out = append(out, g.nodes[id].Key)
	}
	return out, nil
}
```

```go
type Path struct {
	Nodes []NodeID
	Edges []EdgeID
}

func (p Path) Len() int { return len(p.Edges) }

const noParent = ^uint32(0)

// ShortestPathIDs 双向 BFS：小前沿一侧优先整层展开，正反向 visited
// 相交即得最短路径。比单向 BFS 快一个数量级以上（设计文档实测 13x）。
func (g *Graph) ShortestPathIDs(from, to NodeID) (Path, bool, error) {
	if !g.sealed {
		return Path{}, false, ErrNotSealed
	}
	n := uint32(len(g.nodes))
	if uint32(from) >= n || uint32(to) >= n {
		return Path{}, false, ErrNodeNotFound
	}
	if from == to {
		return Path{Nodes: []NodeID{from}}, true, nil
	}
	nInt := int(n)
	fVis := visitedPool.acquire(wordsFor(nInt))
	bVis := visitedPool.acquire(wordsFor(nInt))
	defer visitedPool.release(fVis)
	defer visitedPool.release(bVis)
	fParent := make([]uint32, nInt)
	bParent := make([]uint32, nInt)
	for i := range fParent {
		fParent[i], bParent[i] = noParent, noParent
	}

	fFront := []NodeID{from}
	bFront := []NodeID{to}
	fVis[from>>5] |= 1 << (from & 31)
	bVis[to>>5] |= 1 << (to & 31)

	for len(fFront) > 0 && len(bFront) > 0 {
		// 展开较小一侧（无标签过滤的纯可达层）
		if len(fFront) <= len(bFront) {
			if meet, ok := g.expandMeeting(fFront, fVis, bVis, fParent, g.csrF); ok {
				return g.reconstruct(meet, fParent, bParent), true, nil
			}
			fFront = g.stepFrontier(fFront, fVis, fParent, g.csrF)
		} else {
			if meet, ok := g.expandMeeting(bFront, bVis, fVis, bParent, g.csrR); ok {
				return g.reconstruct(meet, fParent, bParent), true, nil
			}
			bFront = g.stepFrontier(bFront, bVis, bParent, g.csrR)
		}
	}
	return Path{}, false, nil
}

func (g *Graph) ShortestPath(fromKey, toKey string) (Path, bool, error) {
	fn, ok := g.NodeByKey(fromKey)
	if !ok {
		return Path{}, false, ErrNodeNotFound
	}
	tn, ok := g.NodeByKey(toKey)
	if !ok {
		return Path{}, false, ErrNodeNotFound
	}
	return g.ShortestPathIDs(fn.ID, tn.ID)
}

// expandMeeting 在标记本侧 visited 的同时检测对侧 visited。
// 新标记节点 v 若已被对侧访问则记录 meet（返回第一个即可——层同步
// 保证此时相遇距离已最优）。
func (g *Graph) expandMeeting(frontier []NodeID, mine, theirs []uint32, parent []uint32, adj csrAdjacency) (uint32, bool) {
	var meet uint32
	found := false
	for _, u := range frontier {
		for _, v := range adj.Neighbors(uint32(u)) {
			w := &mine[v>>5]
			bit := uint32(1) << (v & 31)
			if *w&bit != 0 {
				continue
			}
			*w |= bit
			parent[v] = uint32(u)
			if !found {
				tb := theirs[v>>5] & bit
				if tb != 0 {
					meet = v
					found = true
				}
			}
		}
	}
	return meet, found
}

func (g *Graph) stepFrontier(frontier []NodeID, visited []uint32, parent []uint32, adj csrAdjacency) []NodeID {
	var next []NodeID
	for _, u := range frontier {
		for _, v := range adj.Neighbors(uint32(u)) {
			w := &visited[v>>5]
			bit := uint32(1) << (v & 31)
			if *w&bit != 0 {
				continue
			}
			*w |= bit
			parent[v] = uint32(u)
			next = append(next, NodeID(v))
		}
	}
	return next
}

// reconstruct 从相遇点向两侧回溯父指针拼接路径。
// 注意 bParent 在反向图上的语义：bParent[v]=u 表示反向图中 u->v，
// 即原图中 v->u —— 回溯得到 meet->to 的正向路径。
func (g *Graph) reconstruct(meet uint32, fParent, bParent []uint32) Path {
	var fwd []NodeID // meet -> from 的逆序
	for v := meet; v != noParent; v = fParent[v] {
		fwd = append(fwd, NodeID(v))
	}
	// 反转得 from -> meet
	for i, j := 0, len(fwd)-1; i < j; i, j = i+1, j-1 {
		fwd[i], fwd[j] = fwd[j], fwd[i]
	}
	var bwd []NodeID // meet -> to
	for v := bParent[meet]; v != noParent; v = bParent[v] {
		bwd = append(bwd, NodeID(v))
	}
	nodes := append(fwd, bwd...)
	edges := make([]EdgeID, 0, len(nodes)-1)
	for i := 0; i+1 < len(nodes); i++ {
		edges = append(edges, g.anyEdge(nodes[i], nodes[i+1]))
	}
	return Path{Nodes: nodes, Edges: edges}
}

// anyEdge 返回 u->v 的任意一条边（多边时任取首条）。
func (g *Graph) anyEdge(u, v NodeID) EdgeID {
	tg, eids, _ := g.csrF.Row(uint32(u))
	for i, t := range tg {
		if t == uint32(v) {
			return EdgeID(eids[i])
		}
	}
	return 0 // 不可达：调用方保证路径连续，正常不会走到
}
```

注意：`expandMeeting` 完整展开当层后才返回 meet，但同层剩余节点仍会继续标记（保证下轮一侧 frontier 完整）。此处 `parent[v]` 用 uint32 存 NodeID。`reconstruct` 中 bwd 从 `bParent[meet]` 起步（meet 不重复）。

```go
type Cycle struct {
	Nodes []NodeID
}

func (c Cycle) Len() int { return len(c.Nodes) }

// Cycles 返回图中所有循环依赖：大小>1 的强连通分量（成员列表）
// 与自环节点。写入侧 SCC 压缩的副产品——"循环依赖阻断"的防御性
// 机制变成"循环依赖发现"的诊断性能力。
func (g *Graph) Cycles() []Cycle {
	if !g.sealed {
		return nil
	}
	var cycles []Cycle
	for s := uint32(0); s < g.sccCount; s++ {
		size := g.sccNodeOff[s+1] - g.sccNodeOff[s]
		if size > 1 {
			mem := g.sccNodes[g.sccNodeOff[s]:g.sccNodeOff[s+1]]
			ids := make([]NodeID, size)
			for i, v := range mem {
				ids[i] = NodeID(v)
			}
			cycles = append(cycles, Cycle{Nodes: ids})
		}
	}
	for u, self := range g.selfLoop {
		if self {
			cycles = append(cycles, Cycle{Nodes: []NodeID{NodeID(u)}})
		}
	}
	slices.SortFunc(cycles, func(a, b Cycle) int {
		if a.Len() != b.Len() {
			return b.Len() - a.Len()
		}
		return int(a.Nodes[0]) - int(b.Nodes[0])
	})
	return cycles
}
```

```go
// 只做崩溃恢复：整图重写、加载后重建 CSR/SCC，无需增量 WAL。
package store

import (
	"bufio"
	"encoding/binary"
	"encoding/json"
	"errors"
	"io"
)

var magic = [4]byte{'V', ' ' != 0: '', '', ''} // 实现时写作 [4]byte{'V','C','T','R'}

const version = 1

type Node struct {
	Key   string
	Label string
	Props map[string]any
}

type Edge struct {
	SrcKey string
	DstKey string
	Label  string
	Props  map[string]any
}

type errShortRead string

func (e errShortRead) Error() string { return string(e) }

func writeString(w *bufio.Writer, s string) error {
	var lenBuf [4]byte
	binary.LittleEndian.PutUint32(lenBuf[:], uint32(len(s)))
	if _, err := w.Write(lenBuf[:]); err != nil {
		return err
	}
	_, err := w.WriteString(s)
	return err
}

func writeProps(w *bufio.Writer, props map[string]any) error {
	if props == nil {
		if err := binary.Write(w, binary.LittleEndian, uint32(0)); err != nil {
			return err
		}
		return nil
	}
	b, err := json.Marshal(props)
	if err != nil {
		return err
	}
	if err := binary.Write(w, binary.LittleEndian, uint32(len(b))); err != nil {
		return err
	}
	_, err = w.Write(b)
	return err
}

func Encode(w io.Writer, nodes []Node, edges []Edge) error {
	bw := bufio.NewWriter(w)
	var header [10]byte
	copy(header[0:4], []byte("VCTR"))
	binary.LittleEndian.PutUint16(header[4:6], version)
	binary.LittleEndian.PutUint32(header[6:10], uint32(len(nodes)))
	if _, err := bw.Write(header[:]); err != nil {
		return err
	}
	if err := binary.Write(bw, binary.LittleEndian, uint32(len(edges))); err != nil {
		return err
	}
	for _, n := range nodes {
		if err := writeString(bw, n.Key); err != nil {
			return err
		}
		if err := writeString(bw, n.Label); err != nil {
			return err
		}
		if err := writeProps(bw, n.Props); err != nil {
			return err
		}
	}
	for _, e := range edges {
		if err := writeString(bw, e.SrcKey); err != nil {
			return err
		}
		if err := writeString(bw, e.DstKey); err != nil {
			return err
		}
		if err := writeString(bw, e.Label); err != nil {
			return err
		}
		if err := writeProps(bw, e.Props); err != nil {
			return err
		}
	}
	return bw.Flush()
}

type reader struct {
	r *bufio.Reader
}

func (rd *reader) u32() (uint32, error) {
	var b [4]byte
	if _, err := io.ReadFull(rd.r, b[:]); err != nil {
		return 0, err
	}
	return binary.LittleEndian.Uint32(b[:]), nil
}

func (rd *reader) str() (string, error) {
	n, err := rd.u32()
	if err != nil {
		return "", err
	}
	b := make([]byte, n)
	if _, err := io.ReadFull(rd.r, b); err != nil {
		return "", err
	}
	return string(b), nil
}

func (rd *reader) props() (map[string]any, error) {
	n, err := rd.u32()
	if err != nil {
		return nil, err
	}
	if n == 0 {
		return nil, nil
	}
	b := make([]byte, n)
	if _, err := io.ReadFull(rd.r, b); err != nil {
		return nil, err
	}
	var m map[string]any
	if err := json.Unmarshal(b, &m); err != nil {
		return nil, err
	}
	return m, nil
}

func Decode(r io.Reader) ([]Node, []Edge, error) {
	rd := &reader{r: bufio.NewReader(r)}
	var m [4]byte
	if _, err := io.ReadFull(rd.r, m[:]); err != nil {
		return nil, nil, err
	}
	if string(m[:]) != "VCTR" {
		return nil, nil, errors.New("store: bad magic")
	}
	var vb [2]byte
	if _, err := io.ReadFull(rd.r, vb[:]); err != nil {
		return nil, nil, err
	}
	if binary.LittleEndian.Uint16(vb[:]) != version {
		return nil, nil, errors.New("store: unsupported version")
	}
	nn, err := rd.u32()
	if err != nil {
		return nil, nil, err
	}
	ne, err := rd.u32()
	if err != nil {
		return nil, nil, err
	}
	nodes := make([]Node, nn)
	for i := range nodes {
		if nodes[i].Key, err = rd.str(); err != nil {
			return nil, nil, err
		}
		if nodes[i].Label, err = rd.str(); err != nil {
			return nil, nil, err
		}
		if nodes[i].Props, err = rd.props(); err != nil {
			return nil, nil, err
		}
	}
	edges := make([]Edge, ne)
	for i := range edges {
		if edges[i].SrcKey, err = rd.str(); err != nil {
			return nil, nil, err
		}
		if edges[i].DstKey, err = rd.str(); err != nil {
			return nil, nil, err
		}
		if edges[i].Label, err = rd.str(); err != nil {
			return nil, nil, err
		}
		if edges[i].Props, err = rd.props(); err != nil {
			return nil, nil, err
		}
	}
	return nodes, edges, nil
}
```

```go
func (g *Graph) Save(w io.Writer) error {
	if !g.sealed {
		return ErrNotSealed
	}
	nodes := make([]store.Node, len(g.nodes))
	for i, n := range g.nodes {
		nodes[i] = store.Node{Key: n.Key, Label: n.Label, Props: n.Props}
	}
	edges := make([]store.Edge, len(g.edges))
	for i, e := range g.edges {

			DstKey: g.nodes[e.dst].Key,
			Label:  g.labels[e.label],
			Props:  e.props,
		}
	}

}
```

```go
package vectra

// Query 是链式遍历构造器：
//
//	g.Query().From("gateway").Out("CALLS").MaxDepth(3).Run()
type Query struct {
	g        *Graph
	fromKeys []string
	labels   []string
	dir      Direction
	maxDepth uint32
}

func (g *Graph) Query() *Query {
	return &Query{g: g, maxDepth: ^uint32(0), dir: DirOut}
}

func (q *Query) From(keys ...string) *Query { q.fromKeys = keys; return q }
func (q *Query) Out(labels ...string) *Query { q.labels, q.dir = labels, DirOut; return q }
func (q *Query) In(labels ...string) *Query  { q.labels, q.dir = labels, DirIn; return q }
func (q *Query) MaxDepth(d uint32) *Query    { q.maxDepth = d; return q }

func (q *Query) Run() (*BFSResult, error) {
	if len(q.fromKeys) == 0 {
		return nil, ErrNodeNotFound
	}
	starts := make([]NodeID, 0, len(q.fromKeys))
	for _, k := range q.fromKeys {
		n, ok := q.g.NodeByKey(k)
		if !ok {
			return nil, ErrNodeNotFound
		}
		starts = append(starts, n.ID)
	}
	opts := []BFSOption{WithDirection(q.dir), WithMaxDepth(q.maxDepth)}
	if len(q.labels) > 0 {
		opts = append(opts, WithLabels(q.labels...))
	}
	return q.g.BFS(starts, opts...)
}

// NeighborsOf 返回 key 节点的一跳邻居（按标签过滤，空 = 全部）。
func (g *Graph) NeighborsOf(key string, dir Direction, labels ...string) ([]Node, error) {
	n, ok := g.NodeByKey(key)
	if !ok {
		return nil, ErrNodeNotFound
	}
	if !g.sealed {
		return nil, ErrNotSealed
	}
	adj := g.csrF
	if dir == DirIn {
		adj = g.csrR
	}
	var mask []bool
	if len(labels) > 0 {
		mask = make([]bool, len(g.labels))
		for _, l := range labels {
			if id, ok := g.labIDs[l]; ok {
				mask[id] = true
			}
		}
	}
	seen := make([]bool, len(g.nodes))
	var out []Node
	tg, lb := adj.Row(uint32(n.ID))
	for i, v := range tg {
		if mask != nil && !mask[lb[i]] {
			continue
		}
		if !seen[v] {
			seen[v] = true
			out = append(out, g.nodes[v])
		}
	}
	return out, nil
}
```

```go
// map 邻接基线：朴素 Go 写法
func bfsMap(adj map[int32][]int32, start int32, maxNodes int32) int {
	visited := make([]bool, maxNodes)
	frontier := []int32{start}
	visited[start] = true
	count := 1
	for len(frontier) > 0 {
		var next []int32
		for _, u := range frontier {
			for _, v := range adj[u] {
				if !visited[v] {
					visited[v] = true
					count++
					next = append(next, v)
				}
			}
		}
		frontier = next
	}
	return count
}

func BenchmarkBFSMap1M(b *testing.B) {
```
